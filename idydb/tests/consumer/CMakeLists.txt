cmake_minimum_required(VERSION 3.16)
option(IDYDB_CONSUMER_C_ONLY "Verify the installed target from a C-only project" OFF)
if(IDYDB_CONSUMER_C_ONLY)
    project(idydb_installed_consumer LANGUAGES C)
else()
    project(idydb_installed_consumer LANGUAGES C CXX)
endif()
find_package(idydb 0.1 CONFIG REQUIRED)
enable_testing()
add_executable(consumer_c consumer.c)
target_link_libraries(consumer_c PRIVATE idydb::idydb)
add_test(NAME consumer_c COMMAND consumer_c)
if(NOT IDYDB_CONSUMER_C_ONLY)
    add_executable(consumer_cpp consumer.cpp)
    target_link_libraries(consumer_cpp PRIVATE idydb::idydb)
    add_test(NAME consumer_cpp COMMAND consumer_cpp)
endif()
